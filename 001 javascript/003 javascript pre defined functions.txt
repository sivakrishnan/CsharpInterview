2. pop() – give the last element in the array

let arr=[3, 4, 1, 7, 5];
let chkPop=arr.pop();
console.log(chkPop); //Output: 5 ---> get the last element in the array
console.log(arr);    // Output: [3, 4, 1, 7]

-------------------------------------------------------------

3. shift() – get first item in the array and remove the first element in that array

let arr=[3, 4, 1, 7, 5];
let chkShift=arr.shift();
console.log(chkShift);   //Output: 3
console.log(arr);		 //Output: [4, 1, 7, 5]

--------------------------------------------------------

4. unshift() – Add item at the beginning

let arr=[3, 4, 1, 7, 5];
let chkUnshift=arr.unshift(2, 6, 9);
console.log(chkUnshift);   //Output: 8
console.log(arr);		   //Output:	[2, 6, 9, 3, 4, 1, 7, 5]  

-----------------------------------------------------------

let arr=[3, 4, 1, 7, 5];

5. find() – Get first matching item (condition based search)
let arr = [10, 20, 30];
let result = arr.find(x=>x>4); // 7 only comes. only first matching

6. findIndex() – Get index of first match - (condition based search)
let arr = [3, 4, 1, 7, 5, 2, 4, 5];
let index = arr.findIndex(x=>x>4); // 3

7. findLastIndex() – Get index of Last match - (condition based search)
let arr = [3, 4, 1, 7, 5, 2, 4, 5, 2]
let chkLastIndex=arr.findLastIndex(x=>x>4);  //short form of callback function
console.log(chkLastIndex);  // 7

7.1 detailed callback function 

let arr=[3, 4, 1, 7, 5, 2, 4, 5, 2];
let chkFind= arr.find(function (x)
                      {
                          return x>4;
                      });
console.log(chkFind);

8. indexOf() - Get index of first match - (exact value based search)
let arr = [3, 4, 1, 7, 5, 2, 4, 5, 2]
let chkIndexOf=arr.indexOf(2);
console.log(chkIndexOf); // 5

9) IndexOf() vs findIndex();

let users=[
    {id:1, name:'siva'}, 
    {id:2, name:'krish'},
    {id: 3, name: 'sivakrish'}
];
let chkIndexOf=users.indexOf({id:2, name:'krish'});
console.log(chkIndexOf);  // -1 not working for object type (exact value based search)
let chkFindIndex=users.findIndex(x=>x.id==2);
console.log(chkFindIndex); // 1 working perfectly for object type (condition based search)

let users=[
    {id:1, name:'siva'},

10. some  – At least one element satisfies condition?

let arr=[3, 4, 1, 7, 5, 2, 4, 5, 2];
let chkSome= arr.some(x=>x>8);
console.log(chkSome); 					 //false
let chkSome1=arr.some(x=>x>5);
console.log(chkSome1); 					//true

11. every() – All elements satisfy condition?

let arr=[3, 4, 1, 7, 5, 2, 4, 5, 2];
let chkEvery= arr.every(x=>x>2);
console.log(chkEvery);             //false
let chkEvery1=arr.every(x=>x>0);
console.log(chkEvery1);				//true

12. includes() – Check if element exists
let arr=[3, 4, 1, 7, 5, 2, 4, 5, 2];

13. map() – Loop through array & return a new array
let arr=[3, 4, 1, 7, 5, 2, 4, 5, 2];
let result=arr.map(x=>x+10);
console.log(result); // [13, 14, 11, 17, 15, 12, 14, 15, 12]

14. filter() – Filter array elements
let arr=[3, 4, 1, 7, 5, 2, 4, 5, 2];
let resFilter=arr.filter(x=>x>4);
console.log(resFilter); // [7, 5, 5]

15. join() – Convert array → string

let users=[
    {id:1, name:'siva'}, 
    {id:2, name:'krish'},
    {id: 3, name: 'sivakrish'}
];

let chkJoin=users.map(x=>x.name).join("-");
console.log(chkJoin);   //siva-krish-sivakrish

11. reduce() – Combine all values into one
let arr = [1, 2, 3];
let sum = arr.reduce((a, b) => a + b, 0); // 6

12. slice() – Copy part of array (non-destructive)
let arr = [1, 2, 3, 4, 5, 6, 7, 8];
let part = arr.slice(2, 4); // [3, 4]

13. splice() – Add/remove elements (destructive)
let arr = [1, 2, 3, 4, 5, 6, 7, 8];
arr.splice(1, 3); // [ 1, 5, 6, 7, 8 ]

14. concat() – Merge arrays
let arr1 = [1, 2];
let arr2 = [3, 4];
let result = arr1.concat(arr2); // [1, 2, 3, 4]

15. sort() – Sort values
let arr = [3, 1, 2];
arr.sort();   // [1, 2, 3]

16. reverse()
let arr = [1, 2, 3];
arr.reverse(); // [3, 2, 1]


20. flat() – Flatten nested arrays

[1, [2, [3]]].flat(2); // [1, 2, 3]
